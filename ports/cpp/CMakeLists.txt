cmake_minimum_required(VERSION 3.20)
project(hqtui_cpp VERSION 0.1.11 LANGUAGES C CXX)
include(CTest)
include(GNUInstallDirs)
if(NOT TARGET hqtui::c)
    add_subdirectory(../c ${CMAKE_CURRENT_BINARY_DIR}/c)
endif()
if(HQTUI_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT hqtui_cpp_ipo_supported OUTPUT hqtui_cpp_ipo_error)
    if(NOT hqtui_cpp_ipo_supported)
        message(FATAL_ERROR "LTO requested but unavailable: ${hqtui_cpp_ipo_error}")
    endif()
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
endif()
add_library(hqtui_cpp INTERFACE)
add_library(hqtui::cpp ALIAS hqtui_cpp)
target_compile_features(hqtui_cpp INTERFACE cxx_std_17)
target_include_directories(hqtui_cpp INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(hqtui_cpp INTERFACE hqtui::c)
set_target_properties(hqtui_cpp PROPERTIES EXPORT_NAME cpp)
if(BUILD_TESTING)
    add_executable(hqtui_cpp_test tests/core.cpp)
    target_link_libraries(hqtui_cpp_test PRIVATE hqtui::cpp)
    add_test(NAME cpp_core COMMAND hqtui_cpp_test)
    add_executable(hqtui_cpp_bench tests/benchmark.cpp)
    target_link_libraries(hqtui_cpp_bench PRIVATE hqtui::cpp)
    add_test(NAME cpp_warm_allocations COMMAND hqtui_cpp_bench 100)
endif()
install(TARGETS hqtui_cpp EXPORT hqtuiTargets)
install(FILES include/hqtui.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
