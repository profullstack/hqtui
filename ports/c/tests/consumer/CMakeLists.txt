cmake_minimum_required(VERSION 3.20)
project(hqtui_consumer LANGUAGES C CXX)
find_package(hqtui CONFIG REQUIRED)
add_executable(consumer_c ../core.c)
target_link_libraries(consumer_c PRIVATE hqtui::c)
add_executable(consumer_cpp ../../../cpp/tests/core.cpp)
target_link_libraries(consumer_cpp PRIVATE hqtui::cpp)
enable_testing()
add_test(NAME installed_c COMMAND consumer_c)
add_test(NAME installed_cpp COMMAND consumer_cpp)
