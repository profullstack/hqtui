cmake_minimum_required(VERSION 3.20)
project(hqtui_c VERSION 0.1.12 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(HQTUI_SANITIZE "Build with address/undefined behavior sanitizers" OFF)
option(HQTUI_LTO "Enable interprocedural optimization when supported" OFF)
if(HQTUI_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT hqtui_ipo_supported OUTPUT hqtui_ipo_error)
    if(NOT hqtui_ipo_supported)
        message(FATAL_ERROR "LTO requested but unavailable: ${hqtui_ipo_error}")
    endif()
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
endif()
add_library(hqtui src/color.c src/unicode.c src/buffer.c src/layout.c src/diff.c src/theme.c src/surface.c)
add_library(hqtui::c ALIAS hqtui)
set_target_properties(hqtui PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO
    VERSION ${PROJECT_VERSION} SOVERSION 0 POSITION_INDEPENDENT_CODE ON EXPORT_NAME c)
target_include_directories(hqtui PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NOT MSVC)
    target_compile_options(hqtui PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    target_link_libraries(hqtui PUBLIC m)
endif()
if(BUILD_TESTING)
    add_executable(hqtui_c_test tests/core.c)
    target_link_libraries(hqtui_c_test PRIVATE hqtui::c)
    add_test(NAME c_core COMMAND hqtui_c_test)
    add_executable(hqtui_c_bench tests/benchmark.c)
    target_link_libraries(hqtui_c_bench PRIVATE hqtui::c)
    add_test(NAME c_warm_allocations COMMAND hqtui_c_bench 100)
    if(BUILD_SHARED_LIBS AND NOT HQTUI_SANITIZE)
        find_package(Python3 COMPONENTS Interpreter)
        if(Python3_Interpreter_FOUND)
            add_test(NAME c_conformance COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/conformance.py $<TARGET_FILE:hqtui>)
        endif()
    endif()
endif()
if(HQTUI_SANITIZE AND NOT MSVC)
    target_compile_options(hqtui PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(hqtui PUBLIC -fsanitize=address,undefined)
endif()
install(TARGETS hqtui EXPORT hqtuiTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/hqtui.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT hqtuiTargets NAMESPACE hqtui:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/hqtui)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/hqtuiConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/hqtuiConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/hqtui)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/hqtuiConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/hqtuiConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/hqtuiConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/hqtui)
file(RELATIVE_PATH HQTUI_PC_PREFIX_RELATIVE "${CMAKE_INSTALL_FULL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_PREFIX}")
configure_file(cmake/hqtui.pc.in ${CMAKE_CURRENT_BINARY_DIR}/hqtui.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/hqtui.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/../../LICENSE DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/licenses/hqtui)
